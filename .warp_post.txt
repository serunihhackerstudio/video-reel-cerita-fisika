/* ============ [2] RENDERER + PASCA-PROSES ============
   Adegan dirender ke target HalfFloat (HDR linear) → bright-pass → blur bertingkat (bloom) → god-ray radial
   → komposit akhir: ACES + gamma + split-tone + tilt-shift + vinyet. Target 8-bit = kotak gelap/banding. */
const REN = new THREE.WebGLRenderer({antialias: false, preserveDrawingBuffer: true, powerPreference: 'high-performance'});
REN.setPixelRatio(1); REN.setSize(W, H); REN.domElement.id = 'gl'; $('#stage').prepend(REN.domElement);
REN.shadowMap.enabled = true; REN.shadowMap.type = THREE.PCFSoftShadowMap;       // tone mapping dikerjakan di shader akhir
const HF = THREE.HalfFloatType, LF = THREE.LinearFilter;
const RT = new THREE.WebGLMultisampleRenderTarget(W, H, {type: HF, minFilter: LF, magFilter: LF}); RT.samples = 4;
const rtBaru = (w, h) => new THREE.WebGLRenderTarget(w | 0, h | 0, {type: HF, minFilter: LF, magFilter: LF, depthBuffer: false});
const R2 = rtBaru(W / 2, H / 2), B4a = rtBaru(W / 4, H / 4), B4b = rtBaru(W / 4, H / 4), B8a = rtBaru(W / 8, H / 8), B8b = rtBaru(W / 8, H / 8);
const RSIN = rtBaru(W / 4, H / 4), RRAY = rtBaru(W / 4, H / 4), RTS = rtBaru(W / 4, H / 4), RTSb = rtBaru(W / 4, H / 4);
const QS = new THREE.Scene(), QC = new THREE.OrthographicCamera(-1, 1, 1, -1, 0, 1), QM = new THREE.Mesh(new THREE.PlaneGeometry(2, 2)); QM.frustumCulled = false; QS.add(QM);
const VSQ = 'varying vec2 vUv;void main(){vUv=uv;gl_Position=vec4(position.xy,0.,1.);}';
const sq = (fs, u) => new THREE.ShaderMaterial({uniforms: u, vertexShader: VSQ, fragmentShader: fs, depthTest: false, depthWrite: false});
const lewat = (mat, tujuan) => { QM.material = mat; REN.setRenderTarget(tujuan); REN.render(QS, QC); };
const pSalin = sq('uniform sampler2D t;varying vec2 vUv;void main(){gl_FragColor=texture2D(t,vUv);}', {t: {value: null}});
const pTerang = sq('uniform sampler2D t;uniform float th;varying vec2 vUv;void main(){vec3 c=max(texture2D(t,vUv).rgb,0.);float l=max(c.r,max(c.g,c.b));gl_FragColor=vec4(c*max(l-th,0.)/max(l,1e-4),1.);}', {t: {value: null}, th: {value: 1}});
const pBlur = sq(`uniform sampler2D t;uniform vec2 d;varying vec2 vUv;void main(){vec3 c=texture2D(t,vUv).rgb*.227;
  c+=(texture2D(t,vUv+d*1.385).rgb+texture2D(t,vUv-d*1.385).rgb)*.316;c+=(texture2D(t,vUv+d*3.231).rgb+texture2D(t,vUv-d*3.231).rgb)*.0703;gl_FragColor=vec4(c,1.);}`, {t: {value: null}, d: {value: new THREE.Vector2()}});
const pSinar = sq(`uniform sampler2D t;uniform vec2 sp;varying vec2 vUv;void main(){vec2 d=(vUv-sp)*.9/48.;vec2 p=vUv;vec3 a=vec3(0.);float w=1.;
  for(int i=0;i<48;i++){p-=d;a+=texture2D(t,clamp(p,0.,1.)).rgb*w;w*=.958;}gl_FragColor=vec4(a/48.,1.);}`, {t: {value: null}, sp: {value: new THREE.Vector2()}});
const AKHIR = sq(`uniform sampler2D tS,tB1,tB2,tR,tT;uniform float ex,bl,ry,tilt,fok,sat,vig,hangat,gelap;varying vec2 vUv;
  vec3 aces(vec3 x){return clamp((x*(2.51*x+.03))/(x*(2.43*x+.59)+.14),0.,1.);}
  void main(){vec2 o=(vUv-.5)*vec2(.0013,.0008);
   vec3 c=vec3(texture2D(tS,vUv+o).r,texture2D(tS,vUv).g,texture2D(tS,vUv-o).b);
   c=mix(c,texture2D(tT,vUv).rgb,tilt*smoothstep(.14,.44,abs(vUv.y-fok)));          // tilt-shift: atas & bawah sedikit kabur
   c+=texture2D(tB1,vUv).rgb*bl*.6+texture2D(tB2,vUv).rgb*bl+texture2D(tR,vUv).rgb*ry;
   c=aces(max(c,0.)*ex);c=pow(c,vec3(1./2.2));
   float l=dot(c,vec3(.299,.587,.114));c=mix(vec3(l),c,sat);
   c*=mix(vec3(1.),vec3(1.05,1.,.92),hangat*l);c=mix(c,c*vec3(.9,1.01,1.07),(1.-l)*.4);   // sorot hangat, bayangan teal
   vec2 d=vUv-.5;c*=mix(1.-vig,1.,smoothstep(.95,.28,length(d*vec2(1.,1.12))));
   gl_FragColor=vec4(clamp(c*(1.-gelap),0.,1.),1.);}`,
  {tS: {value: null}, tB1: {value: null}, tB2: {value: null}, tR: {value: null}, tT: {value: null}, ex: {value: 1}, bl: {value: .55}, ry: {value: 0},
   tilt: {value: .55}, fok: {value: .5}, sat: {value: 1.04}, vig: {value: .45}, hangat: {value: .5}, gelap: {value: 0}});
function blur2(a, b, s) { const w = a.width, h = a.height; pBlur.uniforms.t.value = a.texture; pBlur.uniforms.d.value.set(s / w, 0); lewat(pBlur, b); pBlur.uniforms.t.value = b.texture; pBlur.uniforms.d.value.set(0, s / h); lewat(pBlur, a); }
